
#include <WiFi.h>
#include <WebServer.h>
#include <DHT.h>

#define DHTPIN 27
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
WebServer server(80);

const char* ssid = "ESP32_TEMP";
const char* password = "12345678";

float temperature = 0;
float humidity = 0;
bool readingOK = false;

unsigned long lastRead = 0;

void home() {
  String page = R"rawliteral(
  <!DOCTYPE html>
  <html>
  <head>
    <meta name="viewport"
          content="width=device-width,initial-scale=1">
    <style>
      body {
        font-family: Arial;
        text-align: center;
        background: #f0f4f8;
        padding: 20px;
      }
      .card {
        background: white;
        padding: 25px;
        margin: 15px auto;
        max-width: 300px;
        border-radius: 12px;
        box-shadow: 0 2px 8px #ddd;
      }
      h2 { color: #123b70; }
      .value {
        font-size: 30px;
        color: #16805d;
      }
    </style>
  </head>
  <body>
    <h2>ESP32 Environment Monitor</h2>

    <div class="card">
      <h3>Temperature</h3>
      <div class="value" id="temp">--</div>
    </div>

    <div class="card">
      <h3>Humidity</h3>
      <div class="value" id="hum">--</div>
    </div>

    <p id="status">Reading sensor...</p>

    <script>
      async function update() {
        try {
          const res = await fetch('/readings');
          const data = await res.json();

          if (data.ok) {
            document.getElementById('temp').innerText =
              data.temperature.toFixed(1) + ' °C';

            document.getElementById('hum').innerText =
              data.humidity.toFixed(1) + ' %';

            document.getElementById('status').innerText =
              'Monitoring Active';
          } else {
            document.getElementById('status').innerText =
              'Sensor reading unavailable';
          }
        } catch (e) {
          document.getElementById('status').innerText =
            'Connection Error';
        }
      }

      update();
      setInterval(update, 2500);
    </script>
  </body>
  </html>
  )rawliteral";

  server.send(200, "text/html", page);
}

void readings() {
  String json = "{\"ok\":";
  json += readingOK ? "true" : "false";

  if (readingOK) {
    json += ",\"temperature\":";
    json += String(temperature, 1);

    json += ",\"humidity\":";
    json += String(humidity, 1);
  }

  json += "}";
  server.send(200, "application/json", json);
}

void setup() {
  Serial.begin(115200);
  dht.begin();

  WiFi.mode(WIFI_AP);
  WiFi.softAP(ssid, password);

  server.on("/", home);
  server.on("/readings", readings);
  server.begin();

  Serial.println("Environment Monitor Started");
  Serial.println(WiFi.softAPIP());
}

void loop() {
  server.handleClient();

  if (millis() - lastRead >= 2500) {
    lastRead = millis();

    float h = dht.readHumidity();
    float t = dht.readTemperature();

    if (isnan(h) || isnan(t)) {
      readingOK = false;
      Serial.println("Sensor Read Failed");
    } else {
      temperature = t;
      humidity = h;
      readingOK = true;

      Serial.print("Temperature: ");
      Serial.print(temperature);
      Serial.print(" C | Humidity: ");
      Serial.print(humidity);
      Serial.println(" %");
    }
  }
}
